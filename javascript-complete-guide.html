3) Functions, Scope, Hoisting & Closures

// Function declaration
function add(a, b) { return a + b; }

// Function expression
const multiply = function(a, b) { return a * b; };

// Arrow function
const square = n => n * n;
{
  let city = "Delhi";   // block scope
  const pin = 110001;    // block scope
}

// var is function-scoped and hoisted (avoid in modern JS)
console.log(counter); // undefined due to hoisting
var counter = 5;
// Closure retains access to outer variables
function createCounter() {
  let count = 0;
  return function() {
    count++;
    return count;
  };
}

const next = createCounter();
console.log(next()); // 1
console.log(next()); // 2

4) Objects & Arrays

Object Use Case: User Profile

const user = {
  id: 101,

  isPremium: true,

    return `Welcome, ${this.name}`;
  }
};

console.log(user.greet());

Array Use Case: Order Calculations

const orders = [500, 1200, 300, 700];
const total = orders.reduce((sum, value) => sum + value, 0);
const bigOrders = orders.filter(x => x >= 700);

console.log(total);      // 2700
console.log(bigOrders);  // [1200, 700]

6) Async JavaScript (Callbacks → Promises → Async/Await)

Promises

function fetchUser() {
  return new Promise(resolve => {
    setTimeout(() => resolve({ id: 1, name: "Aarav" }), 1000);
  });
}

Async/Await (real API call)

async function loadPosts() {
  try {
    const res = await fetch("https://jsonplaceholder.typicode.com/posts?_limit=2");
    const posts = await res.json();
    console.log(posts);
  } catch (err) {
    console.error("Failed to fetch posts", err);
  }
}

8) Advanced Concepts

const team = {
  name: "Core JS",
  show() {
    console.log(this.name);
  }
};

const detached = team.show;
detached(); // undefined in strict mode
const fixed = detached.bind(team);
fixed();    // Core JS
const animal = { eats: true };
const dog = Object.create(animal);
dog.barks = true;

console.log(dog.eats);  // inherited via prototype chain
// Currying
const multiply = a => b => a * b;
const double = multiply(2);
console.log(double(10)); // 20

// Memoization
function memoize(fn) {
  const cache = {};
  return (...args) => {
    const key = JSON.stringify(args);
    if (cache[key]) return cache[key];
    cache[key] = fn(...args);
    return cache[key];
  };
}
console.log("Start");
setTimeout(() => console.log("Timeout"), 0);
Promise.resolve().then(() => console.log("Promise"));
console.log("End");

// Output:
// Start
// End
// Promise
// Timeout

9) Glossary

Closure

A function with access to variables from its lexical scope even after the outer function returns.

Hoisting

JavaScript moves declarations to top of scope during compilation phase.

Promise

An object representing eventual completion/failure of async work.

Event Loop

Mechanism that manages execution stack, callback queue, and microtasks.

Prototype

Object from which another object inherits properties and methods.

Destructuring

Syntax to extract values from arrays/objects into variables.

Pure Function

Function with no side effects and same output for same input.

Debouncing

Technique to limit function execution rate by delaying invocation.
